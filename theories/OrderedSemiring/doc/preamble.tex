    "[" .. mathfont .. "]:mode=node;",
    "[" .. symbols .. "]:mode=node;"
  })
}

% Explicit absolute file-link annotations work even in PDF readers that ignore
% the document URI base. Keep the original renderer for local label references.
\ExplSyntaxOn
\cs_new_eq:NN \ordered_original_link:nnnn \markdownRendererLinkPrototype
\cs_gset_protected:Npn \markdownRendererLinkPrototype #1#2#3#4
  {
    \tl_set:Nn \l_tmpa_tl { #3 }
    \tl_set:Nx \l_tmpb_tl { \tl_range:Nnn \l_tmpa_tl { 1 } { 1 } }
    \str_if_eq:NNTF \l_tmpb_tl \c_hash_str
      { \ordered_original_link:nnnn { #1 } { #2 } { #3 } { #4 } }
      {
        \regex_match:nnTF { \A [A-Za-z] [A-Za-z0-9+.\-]* : } { #3 }
          { \href { #3 } { #1 } }
          { \href { \OrderedSourceBase #3 } { #1 } }
      }
  }
\ExplSyntaxOff
