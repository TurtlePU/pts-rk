\section{Мотивация и идея}
\begin{itemize}
\item Больше всего вопросов при изучении современных
    средств интерактивных доказательств (Lean, Rocq,
    Agda...) вызывает устройство \textit{иерархий
    вселенных}: $\mathtt{Type}_1 : \mathtt{Type}_2 :
    \ldots$
\item Вызывают они вопросы и с метатеоретической точки
    зрения --- в доказательствах здравости системы
    $\mathtt{Type}_n$ интерпретируется как $n$-й
    недостижимый кардинал.
\item Поэтому интересно искать альтернативные варианты
    устройства иерархии. Далее описан забавный вариант,
    располагающий типы по ``вселенным'' соответственно
    \textit{рангу функций}.
\item Б\'{о}льшая часть описанных результатов формально
    верифицирована на Rocq.
\end{itemize}

\subsection{О здравости ЧСТ}
Какие ЧСТ здравы?
\begin{itemize}
    \item Исчисление конструкций:
        \[\begin{array}{c}
            S=\{*,\square\}
            \quad \mathrm{Ax}=\{(*,\square)\}
            \\ \mathrm{Rule}=\{
                (*,*,*),(*,\square,\square),
                (\square,*,*),(\square,\square,\square)
            \}
        \end{array}\]
    \item Первая счётная иерархия \textit{вселенных}:
        \[\begin{array}{c}
            S=\{\mathcal{U}_\ell\;|\;\ell\in\mathbb{N}\}
            \quad \mathrm{Ax}=\{
                (\mathcal{U}_\ell,\mathcal{U}_{\ell+1})
                \;|\;\ell\in\mathbb{N}
            \}
            \\ \mathrm{Rule}=\{
                (\mathcal{U}_\ell, \mathcal{U}_{\ell'},
                \mathcal{U}_{\max(\ell,\ell')}) \;|\;
                \ell,\ell'\in\mathbb{N}
            \}
        \end{array}\]
    \item Счётная иерархия с импредикативным
        \texttt{Prop}:
        \[\begin{array}{rcl}
            S &=&\{\mathtt{Prop}\}\sqcup
            \{\mathcal{U}_\ell\;|\;\ell\in\mathbb{N}\}
            \\ \mathrm{Ax} &=&
            \{(\mathtt{Prop},\mathcal{U}_0)
            \}\sqcup\{
                (\mathcal{U}_\ell,\mathcal{U}_{\ell+1})
                \;|\;\ell\in\mathbb{N}
            \}
            \\ \mathrm{Rule} &=&\{
                (\mathcal{U}_\ell,\mathtt{Prop},
                \mathtt{Prop}) \;|\;
                \ell\in\mathbb{N}
            \} \;\sqcup
            \\ & \sqcup & \{
                (\mathcal{U}_\ell, \mathcal{U}_{\ell'},
                \mathcal{U}_{\max(\ell,\ell')}) \;|\;
                \ell,\ell'\in\mathbb{N}
            \}
        \end{array}\]
\end{itemize}
\subsection{Парадокс Жирара}
    \begin{theorem}[Girard'72]
    Следующая ЧСТ (известная как System U) содержит
    \textbf{не нормализующийся} терм:
    \[\begin{array}{c}
        S=\{*,\square,\triangle\}
        \quad\mathrm{Ax}=\{
            (*,\square),(\square,\triangle)
        \}
        \\ \mathrm{Rule}=\{
            (*,*,*),(\square,*,*),(\triangle,*,*),
            (\square,\square,\square),
            (\triangle,\square,\square)
        \}
    \end{array}\]
    Более того, этот терм имеет тип $\Pi(T:*).T$.
    \end{theorem}
    Другими словами, <<здравая ЧСТ не может поддерживать
    одновременно \textbf{и} импредикативный
    \texttt{Prop}, \textbf{и} полиморфные функции>>.

    Из парадокса Жирара выводится противоречивость ещё
    более простой ЧСТ, известной как Type-in-Type:
    \[S=\{*\}\quad\mathrm{Ax}=\{(*,*)\}\quad
    \mathrm{Rule}=\{(*,*,*)\}\]
\subsection{Как насчёт рангов?}
\textit{Рангом} функции называется максимальная
вложенность $\Pi$ в позиции аргумента:
\[\mathrm{rk}\,(\Pi(x:T).U)=\max\,(\mathrm{rk}\,T+1,
    \mathrm{rk}\,U)\]
Рассмотрим $S=\{\mathrm{rk}_n\;|\;n\in\mathbb{N}\}$, и
\[\mathrm{Rule}=\{(\mathrm{rk}_m,\mathrm{rk}_n,
    \mathrm{rk}_{\max(m+1,n)})\;|\;m,n\in\mathbb{N}\}.\]
В качестве $\mathrm{Ax}$ можно взять цепочку
$\mathrm{rk}_0:\mathrm{rk}_1:\mathrm{rk}_2:\ldots$, но
получится система, строго менее удобная, чем обычная
иерархия вселенных. Попробуем поступить следующим
образом:
\[\mathrm{Ax}=\{(\mathrm{rk}_n,\mathrm{rk}_0)\;|\;
    n\in\mathbb{N}\}\]
В частности, $\mathrm{rk}_0:\mathrm{rk}_0$. Но в данной
системе парадокс Жирара не применим!

\section{Файл \texttt{AbstractRewriting.v}}
%==============================================================================

В данном файле развиваются основные механизмы абстрактного переписывания
над произвольным отношением на множестве.

Файл импортирует библиотеку надмножеств (\emph{setoid library}) из
стандартной библиотеки.

\begin{definition}[Рефлексивно-транзитивная обёртка]
Для множества $A$ и отношения $R$ на $A$ \emph{рефлексивно-транзитивная
обёртка} $\RTC$ отношения $R$ есть наименьшее отношение на $A$,
генерируемое двумя конструкторами: конструктор рефлексивности, дающий
$x\;\RTC\;x$ для каждого $x \in A$; и конструктор шага, который, по
заданным $x\;R\;y$ и $y\;\RTC\;z$, даёт $x\;\RTC\;z$.
\end{definition}

\begin{definition}[Симметрическая обёртка]
\emph{Симметрическая обёртка} $\Sym$ отношения $R$ есть наименьшее
отношение на $A$, генерируемое: \emph{прямым} конструктором, который, по
заданному $x\;R\;y$, даёт $x\;\Sym\;y$; и \emph{обратным} конструктором,
который, по заданному $y\;R\;x$, даёт $x\;\Sym\;y$.
\end{definition}

\begin{definition}[Обратное отношение]
\emph{Обратное} отношение $R^{-1}$ определяется так:
$x\;R^{-1}\;y$ тогда и только тогда, когда $y\;R\;x$.
\end{definition}

\begin{definition}[Эквивалентная обёртка]
\emph{Эквивалентная обёртка} отношения $R$ есть рефлексивно-транзитивная
обёртка симметрической обёртки:
$\EquivClosure R \;\triangleq\; \RTC(\Sym R)$.
\end{definition}

\begin{definition}[Прообраз отношения]
Для функции $f : A \to B$ и отношения $R$ на $B$ \emph{прообраз}
отношения $R$ по $f$ (обозначение $R \mathbin{@} f$) есть отношение на $A$,
определённое тем, что
$x\;(R \mathbin{@} f)\;y$ тогда и только тогда, когда $f(x)\;R\;f(y)$.
\end{definition}

\begin{definition}[Свойство Церча--Россера]
Отношение $R$ на $A$ обладает \emph{свойством Церча--Россера},
если для всех $t, t_1, t_2 \in A$, из $t\;R\;t_1$ и $t\;R\;t_2$
существует $u$ такой, что $t_1\;R\;u$ и $t_2\;R\;u$.
\end{definition}

\begin{instance}
Рефлексивно-транзитивная обёртка любого отношения рефлексивна.
\end{instance}

\begin{instance}
Рефлексивно-транзитивная обёртка любого отношения транзитивна.
\end{instance}

\begin{lemma}
Если $x\;R\;y$, то $x\;\RTC\;y$.
\end{lemma}

\begin{instance}
Если $R$ симметрично, то его рефлексивно-транзитивная обёртка
также симметрична.
\end{instance}

\begin{instance}
Симметрическая обёртка любого отношения симметрична.
\end{instance}

\begin{instance}
Эквивалентная обёртка любого отношения есть отношение эквивалентности
(т.\,е.~рефлексивна, симметрична и транзитивна).
\end{instance}

\begin{lemma}
Если $x\;\RTC(R^{-1})\;y$, то $y\;\RTC(R)\;x$.
\end{lemma}

\begin{lemma}[Отображение через рефлексивно-транзитивную обёртку]
Для функции $f : A \to B$ и отношения $R$ на $B$: если
$x\;\RTC(R \mathbin{@} f)\;y$, то $f(x)\;\RTC(R)\;f(y)$.
\end{lemma}

\begin{lemma}[Монотонность рефлексивно-транзитивной обёртки]
Если $Q \subseteq R$ (т.\,е.~$Q\,x\,y \Rightarrow R\,x\,y$ для всех
$x, y$), то $\RTC Q \subseteq \RTC R$.
\end{lemma}

\begin{lemma}[Отображение через симметрическую обёртку]
Для функции $f : A \to B$ и отношения $R$ на $B$: если
$x\;\Sym(R \mathbin{@} f)\;y$, то $f(x)\;\Sym(R)\;f(y)$.
\end{lemma}

\begin{lemma}[Монотонность симметрической обёртки]
Если $Q \subseteq R$, то $\Sym Q \subseteq \Sym R$.
\end{lemma}

\begin{lemma}
Если $x\;R\;y$, то $x\;\EquivClosure(R)\;y$.
\end{lemma}

\begin{lemma}[Отображение через эквивалентную обёртку]
Для функции $f : A \to B$ и отношения $R$ на $B$: если
$x\;\EquivClosure(R \mathbin{@} f)\;y$, то
$f(x)\;\EquivClosure(R)\;f(y)$.
\end{lemma}

\begin{lemma}[Монотонность эквивалентной обёртки]
Если $Q \subseteq R$, то
$\EquivClosure(Q) \subseteq \EquivClosure(R)$.
\end{lemma}

\begin{lemma}[Экстенциональность свойства Церча--Россера]
Если два отношения $Q$ и $R$ на $A$ равны
(т.\,е.~$Q(a,b) \Leftrightarrow R(a,b)$ для всех $a, b$), то $Q$
обладает свойством Церча--Россера тогда и только тогда, когда $R$
обладает свойством Церча--Россера.
\end{lemma}

\begin{theorem}[Свойство Церча--Россера для рефлексивно-транзитивной обёртки]
Если отношение $R$ обладает свойством Церча--Россера, то
его рефлексивно-транзитивная обёртка $\RTC R$ также обладает
свойством Церча--Россера.
\end{theorem}

\section{Файл \texttt{Syntax.v}}
%==============================================================================

В данном файле определяется синтаксис зависимой теории типов с одним
сортом типов, произведениями, лямбдами, приложениями и переменными,
а также базовые операции манипуляции термами.

Файл импортирует модуль проверенного равенства из стандартной
библиотеки.

\begin{definition}[Ограниченные натуральные числа]
Тип $\Fin(n)$ состоит из натуральных чисел, строго меньших $n$,
с конструкторами $\mathsf{fzero}$ (первый элемент) и $\mathsf{fsucc}$
(преемник элемента).
\end{definition}

\begin{definition}[Термы]
Для множества уровней $L$ и натурального числа $n$ (количество
переменных в контексте) \emph{терм} $\Term(L, n)$ есть одно из:
\emph{вселенная} $\UU(\ell)$ для уровня $\ell \in L$;
\emph{произведение} $\PiT\,T\;U$, где $T$ --- терм в контексте
длины $n$, а $U$ --- терм в контексте длины $n+1$;
\emph{переменная} $\var(i)$ для индекса де~Брюйна
$i \in \Fin(n)$;
\emph{лямбда} $\lam\,T\;t$, где $T$ --- терм в контексте
длины $n$, а $t$ --- терм в контексте длины $n+1$;
или \emph{применение} $f\;\app\;u$ двух термов в контексте
длины $n$.
\end{definition}

\begin{definition}[Матчинг по $\Fin$]
Вспомогательная функция, которая, по заданному значению $z$ для
базового случая и функции $s$ для индуктивного случая,
производит матчинг по индексу де~Брюйна.
\end{definition}

\begin{definition}[Ослабление функции переименования]
Для функции переименования $f : \Fin(m) \to \Fin(n)$
\emph{ослабление} $\weak(f) : \Fin(S\,m) \to \Fin(S\,n)$
отображает новый индекс $\mathsf{fzero}$ на $\mathsf{fzero}$
и сдвигает остальные.
\end{definition}

\begin{definition}[Переименование]
Для функции переименования $f : \Fin(m) \to \Fin(n)$
операция \emph{переименования} преобразует терм в контексте
длины $m$ в терм в контексте длины $n$: вселенные остаются
без изменений; в произведении область переименовывается по $f$,
а тело --- по $\weak(f)$; индексы переменных отображаются по $f$;
в лямбде область переименовывается по $f$, а тело --- по
$\weak(f)$; в применении обе части переименовываются по $f$.
\end{definition}

\begin{definition}[Сдвиг]
\emph{Сдвиг} терма $t$ в контексте длины $n$ даёт терм в контексте
длины $n+1$ путём увеличения каждого индекса переменной
(т.\,е.~переименование по функции, отображающей каждый индекс
на его преемник).
\end{definition}

\begin{definition}[Транспонирование]
Для функции $f : \Fin(m) \to \Term(L, n)$
\emph{транспонирование} по индексу $i$ даёт терм в контексте
длины $n+1$: при $i = \mathsf{fzero}$ это переменная
$\var(\mathsf{fzero})$; при $i = \mathsf{fsucc}(j)$ это
$\shift(f(j))$.
\end{definition}

\begin{definition}[Подстановка]
Для функции $f : \Fin(m) \to \Term(L, n)$ и терма $t$ в контексте
длины $m$ операция \emph{подстановки} заменяет переменные в $t$
соответствующими термами из $f$: вселенные остаются без изменений;
в произведении область подставляется по $f$, а тело --- по
транспонированию $f$; переменная $i$ заменяется на $f(i)$;
в лямбде область подставляется по $f$, а тело --- по транспонированию
$f$; в применении обе части подставляются по $f$.
\end{definition}

\begin{definition}[Подстановка в свежую переменную]
Подстановка терма $u$ в свежую переменную терма $t$
(обозначение $t\;\sub\;u$) есть операция замены переменной
с индексом $\mathsf{fzero}$ на $u$ при сохранении остальных
переменных со сдвигом.
\end{definition}

\begin{definition}[Толкание]
Для терма $t$ в контексте длины $n$ функция \emph{толкания}
производит, для каждого индекса $i$ в $\Fin(m+n)$, соответствующий
терм в контексте длины $m+n$ с надлежащим сдвигом.
\end{definition}

\begin{definition}[Индексная подстановка]
Подстановка терма $u$ в переменную с заданным индексом в терме $t$
(обозначение $t\;\sub^i\;u$) выполняется путём толкания $u$
к нужному индексу и последующего применения подстановки.
\end{definition}

\begin{lemma}[Экстенциональность переименования]
Если две функции переименования $f$ и $g$ совпадают на всех
индексах, то они дают одинаковый результат при применении
к любому терму.
\end{lemma}

\begin{lemma}[Экстенциональность транспонирования]
Если две функции $f$ и $g$ совпадают на всех индексах,
то их транспонирования совпадают на всех индексах.
\end{lemma}

\begin{lemma}[Экстенциональность подстановки]
Если две функции подстановки $f$ и $g$ совпадают на всех
индексах, то они дают одинаковый результат при применении
к любому терму.
\end{lemma}

\begin{lemma}[Композиция переименований]
Переименование по $f$ после переименования по $g$ равносильно
переименованию по композиции $f \circ g$.
\end{lemma}

\begin{lemma}[Переименование коммутирует с подстановкой]
Переименование подставленного терма по $f$ равносильно
подстановке по переименованной функции с последующим применением
результата.
\end{lemma}

\begin{lemma}[Подстановка коммутирует с переименованием]
Подстановка в переименованный терм равносильна подстановке
по композиции функций.
\end{lemma}

\begin{lemma}[Подстановка с тождественным отображением]
Замена переменных на самих себя есть тождественная операция
на термах.
\end{lemma}

\begin{lemma}[Композиция подстановок]
Подстановка одной подстановки в другую равносильна подстановке
по композиции функций.
\end{lemma}

\begin{lemma}[Переименование распределяется над подстановкой]
Переименование подстановки $t\;\sub\;u$ по $f$ равносильно
подстановке $\weak(f)(t)\;\sub\;f(u)$.
\end{lemma}

\begin{lemma}[Подстановка распределяется над подстановкой]
Подстановка в подстановку $t\;\sub\;u$ по $f$ равносильна
подстановке $\transpose(f)(t)\;\sub\;f(u)$.
\end{lemma}

\begin{lemma}[Сдвиг, затем подстановка, --- тождественно]
Сдвиг терма и последующая подстановка переменной в свежий
индекс восстанавливают исходный терм.
\end{lemma}

\begin{lemma}[Сдвиг коммутирует с индексной подстановкой]
Сдвиг, за которым следует индексная подстановка, равносилен
индексной подстановке, за которой следует сдвиг.
\end{lemma}

\section{Файл \texttt{Congruence.v}}
%==============================================================================

В данном файле определяется понятие конгруэнтного отношения
на синтаксисе термов и доказано, что оно сохраняется
при симметрической и рефлексивно-транзитивной обёртках.

Файл импортирует библиотеку надмножеств из стандартной библиотеки,
а также модули синтаксиса и абстрактного переписывания.

\begin{class}[Конгруэнтность]
\emph{Конгруэнтность} для семейства отношений $R$ на термах
(индексируемого длиной контекста) --- это набор из шести
условий совместимости: отношение совместимо с производными
по левому аргументу (если $T \mathrel{R} T'$, то
$\PiT\,T\;U \mathrel{R} \PiT\,T'\;U$) и по правому
(если $U \mathrel{R} U'$, то $\PiT\,T\;U \mathrel{R} \PiT\,T\;U'$);
совместимо с лямбдами по левому аргументу (если
$T \mathrel{R} T'$, то $\lam\,T\;t \mathrel{R} \lam\,T'\;t$)
и по правому (если $t \mathrel{R} t'$, то
$\lam\,T\;t \mathrel{R} \lam\,T\;t'$); совместимо
с приложениями по левому аргументу (если
$t \mathrel{R} t'$, то $t\;\app\;u \mathrel{R} t'\;\app\;u$)
и по правому (если $u \mathrel{R} u'$, то
$t\;\app\;u \mathrel{R} t\;\app\;u'$).
\end{class}

\begin{instance}[Симметрическая обёртка сохраняет конгруэнтность]
Если $R$ --- конгруэнтность, то семейство симметрических
обёрток $\Sym(R)$ также является конгруэнтностью.
\end{instance}

\begin{instance}[Рефлексивно-транзитивная обёртка сохраняет конгруэнтность]
Если $R$ --- конгруэнтность, то семейство рефлексивно-транзитивных
обёрток $\RTC(R)$ также является конгруэнтностью.
\end{instance}

\begin{lemma}[Параллельная конгруэнтность для произведений]
Если $R$ --- транзитивная конгруэнтность, и $T \mathrel{R} T'$,
$U \mathrel{R} U'$, то $\PiT\,T\;U \mathrel{R} \PiT\,T'\;U'$.
\end{lemma}

\begin{lemma}[Параллельная конгруэнтность для лямбд]
Если $R$ --- транзитивная конгруэнтность, и $T \mathrel{R} T'$,
$t \mathrel{R} t'$, то $\lam\,T\;t \mathrel{R} \lam\,T'\;t'$.
\end{lemma}

\begin{lemma}[Параллельная конгруэнтность для приложений]
Если $R$ --- транзитивная конгруэнтность, и $t \mathrel{R} t'$,
$u \mathrel{R} u'$, то $t\;\app\;u \mathrel{R} t'\;\app\;u'$.
\end{lemma}

\section{Файл \texttt{Reduction.v}}
%==============================================================================

В данном файле определяется одношаговое $\beta$-преобразование
и доказаны его базовые свойства.

Файл импортирует модули отношений и проверенного равенства
из стандартной библиотеки, а также модули синтаксиса,
конгруэнтности и абстрактного переписывания.

\begin{definition}[Одношаговое $\beta$-преобразование]
Отношение $\bstep$ на термах в заданном контексте определяется
семью конструкторами: преобразование в области произведения
($\PiT\,T\;U \to \PiT\,T'\;U$, если $T \to T'$);
преобразование в теле произведения ($\PiT\,T\;U \to \PiT\,T\;U'$,
если $U \to U'$); преобразование в области лямбды
($\lam\,T\;t \to \lam\,T'\;t$, если $T \to T'$);
преобразование в теле лямбды ($\lam\,T\;t \to \lam\,T\;t'$,
если $t \to t'$); преобразование в функции приложения
($f\;\app\;t \to f'\;\app\;t$, если $f \to f'$);
преобразование в аргументе приложения ($f\;\app\;t \to f\;\app\;t'$,
если $t \to t'$); и $\beta$-правило
($\lam\,T\;f\;\app\;t \to f\;\sub\;t$).
\end{definition}

\begin{instance}
Одношаговое $\beta$-преобразование является конгруэнтностью.
\end{instance}

\begin{definition}[Многошаговое $\beta$-преобразование]
Отношение $\mstep$ есть рефлексивно-транзитивная обёртка
отношения $\bstep$.
\end{definition}

\begin{lemma}[Переименование коммутирует с шагом]
Если $t \bstep t'$, то
$\rename(f)(t) \bstep \rename(f)(t')$.
\end{lemma}

\begin{lemma}[Подстановка коммутирует с шагом]
Если $t \bstep t'$, то
$\replace(f)(t) \bstep \replace(f)(t')$.
\end{lemma}

\begin{lemma}[Лямбда-инверсия для переименования]
Если лямбда-терм равен переименованию некоторого терма,
то исходный терм сам является лямбдой, и переименование
раскладывается соответственно.
\end{lemma}

\begin{lemma}[Инверсия шага для переименования]
Если $\rename(f)(t) \bstep u$, то существует $v$ такой,
что $u = \rename(f)(v)$ и $t \bstep v$.
\end{lemma}

\begin{lemma}[Переименование коммутирует с многошаговым преобразованием]
Если $t \mstep t'$, то
$\rename(f)(t) \mstep \rename(f)(t')$.
\end{lemma}

\begin{lemma}[Транспонирование коммутирует с многошаговым преобразованием]
Если каждый компонент функции подстановки преобразуется
в соответствующий компонент другой, то их транспонирования
также преобразуются друг в друга.
\end{lemma}

\begin{lemma}[Подстановка коммутирует с многошаговым преобразованием]
Если каждый компонент функции подстановки преобразуется
в соответствующий компонент другой, то подставленные термы
также преобразуются друг в друга.
\end{lemma}

\begin{lemma}[Представление произведения]
Если $\PiT\,T\;U \mstep t$, то $t$ является произведением.
\end{lemma}

\begin{lemma}[Инверсия произведения для многошагового преобразования]
Если $\PiT\,T\;U \mstep \PiT\,T'\;U'$, то
$T \mstep T'$ и $U \mstep U'$.
\end{lemma}

\section{Файл \texttt{Confluence.v}}
%==============================================================================

В данном файле определяется отношение параллельного
$\beta$-преобразования и доказывается свойство связности
(церх--россер) как для параллельного, так и для стандартного
$\beta$-преобразования.

Файл импортирует стандартную библиотеку, модули абстрактного
переписывания, синтаксиса, редукции и конгруэнтности.

\begin{definition}[Параллельное $\beta$-преобразование]
Отношение параллельной редукции $\pstep$ определяется:
рефлексивным шагом ($t \pstep t$); параллельным
преобразованием в области и теле произведения; параллельным
преобразованием в области и теле лямбды; параллельным
преобразованием в функции и аргументе приложения;
и $\beta$-шагом, в котором функция и аргумент могут
содержать внутренние параллельные преобразования
($\lam\,T\;f\;\app\;t \pstep f'\;\sub\;t'$,
если $f \pstep f'$ и $t \pstep t'$).
\end{definition}

\begin{lemma}[Один шаг --- параллельный шаг]
Если $t \bstep t'$, то $t \pstep t'$.
\end{lemma}

\begin{lemma}[Параллельный шаг --- многошаговое преобразование]
Если $t \pstep t'$, то $t \mstep t'$.
\end{lemma}

\begin{lemma}[Переименование коммутирует с параллельным шагом]
Если $t \pstep t'$, то
$\rename(f)(t) \pstep \rename(f)(t')$.
\end{lemma}

\begin{lemma}[Транспонирование коммутирует с параллельным шагом]
Если каждый компонент функции подстановки делает параллельный
шаг в соответствующий компонент другой, то их транспонирования
также делают параллельный шаг.
\end{lemma}

\begin{lemma}[Подстановка коммутирует с параллельным шагом]
Если каждый компонент функции подстановки делает параллельный
шаг в соответствующий компонент другой, и $t \pstep t'$,
то подставленные термы также делают параллельный шаг
друг в друга.
\end{lemma}

\begin{lemma}[Подстановка распределяется над параллельным шагом]
Если $t \pstep t'$, то
$f\;\sub\;t \pstep f\;\sub\;t'$.
\end{lemma}

\begin{lemma}[Параллельные шаги в функции и аргументе]
Если $f \pstep f'$ и $t \pstep t'$, то
$f\;\sub\;t \pstep f'\;\sub\;t'$.
\end{lemma}

\begin{lemma}[Лямбда-инверсия для параллельного шага]
Если лямбда-терм делает параллельный шаг в некоторый терм $t'$,
то $t'$ является лямбдой, и параллельный шаг раскладывается
на параллельные шаги в области и теле.
\end{lemma}

\begin{lemma}[Двойниковое свойство для параллельной редукции]
Отношение параллельного $\beta$-преобразования обладает
свойством Церча--Россера.
\end{lemma}

\begin{theorem}[Связность $\beta$-преобразования]
Отношение стандартного $\beta$-преобразования обладает
свойством Церча--Россера (т.\,е.~$\beta$-преобразование
связно).
\end{theorem}

\section{Файл \texttt{DefEq.v}}

В данном файле определяется тождественное равенство

Файл импортирует стандартную библиотеку, модули абстрактного

\begin{definition}[Тождественное равенство]
Два терма $t$ и $t'$ \emph{равны тождественно}
(обозначение $t \Deq t'$), если они связаны эквивалентной
обёрткой отношения одношагового $\beta$-преобразования.
\end{definition}

\begin{theorem}[Характеризация тождественного равенства через Церча--Россера]
Два терма равны тождественно тогда и только тогда, когда
существует общий результат редукции: т.\,е.~$t \Deq t'$
тогда и только тогда, когда существует терм $u$ такой,
что $t \mstep u$ и $t' \mstep u$.
\end{theorem}

\begin{lemma}[Инверсия произведения для тождественного равенства]
Если $\PiT\,T\;U \Deq \PiT\,T'\;U'$, то
$T \Deq T'$ и $U \Deq U'$.
\end{lemma}

\begin{lemma}[Подстановка сохраняет тождественное равенство]
Если $t \Deq t'$, то
$f\;\sub\;t \Deq f\;\sub\;t'$.
\end{lemma}

\begin{lemma}[Переименование сохраняет тождественное равенство]
Если $t \Deq t'$, то
$\rename(f)(t) \Deq \rename(f)(t')$.
\end{lemma}

\begin{lemma}[Подстановка сохраняет тождественное равенство]
Если $t \Deq t'$, то
$\replace(f)(t) \Deq \replace(f)(t')$.
\end{lemma}

\section{Файл \texttt{Context.v}}
%==============================================================================

В данном файле определяется синтаксис контекстов,
отношение $\beta$-преобразования на контекстах
и вспомогательные операции манипуляции контекстами.

Файл импортирует модули проверенного равенства и отношений
из стандартной библиотеки, а также модули синтаксиса и редукции.

\begin{definition}[Контексты]
\emph{Контекст} длины $n$ --- это либо пустой контекст
$\varepsilon$ (длины $0$), либо расширение
$\Gamma \;\&\; T$ контекста $\Gamma$ длины $n$ термом $T$
в контексте длины $n$, дающее контекст длины $n+1$.
\end{definition}

\begin{definition}[$\beta$-преобразование контекстов]
Отношение $\dstep$ на контекстах определяется:
$\beta$-преобразованием в начале контекста
(если $T \bstep U$, то $\Gamma\;\&\;T \dstep \Gamma\;\&\;U$);
и $\beta$-преобразованием глубже в контексте
(если $\Gamma \dstep \Delta$, то
$\Gamma\;\&\;T \dstep \Delta\;\&\;T$).
\end{definition}

\begin{definition}[Предшественник контекста]
Для непустого контекста $\Gamma = \Gamma_0\;\&\;T$
\emph{предшественник} есть $\Gamma_0$.
\end{definition}

\begin{definition}[Вершина контекста]
Для непустого контекста $\Gamma = \Gamma_0\;\&\;T$
\emph{вершина} (или головной элемент) есть $T$.
\end{definition}

\begin{definition}[Индексирование контекста]
Для контекста $\Gamma$ длины $n$ и индекса
$i \in \Fin(n)$ обозначение $\Gamma \mathbin{!!} i$
извлекает $i$-й тип из контекста (считая с конца),
с надлежащим сдвигом.
\end{definition}

\begin{definition}[Переименователь ослабления]
Функция $\wrr : \Fin(m+n) \to \Fin(S\,(m+n))$,
сдвигающая индексы с учётом нового типа, вставленного
в начало контекста.
\end{definition}

\begin{definition}[Вставка в контекст]
Вставка типа $T$ в начало контекста $\Gamma$ длины $m+n$
даёт контекст длины $S\,(m+n)$ с надлежащим переименованием
существующих типов.
\end{definition}

\begin{definition}[Сжатие контекста]
Удаление первых $m$ типов из контекста длины $m+n$,
оставляя контекст длины $n$.
\end{definition}

\begin{definition}[Индексная подстановка в контекст]
Подстановка терма $t$ в переменную с заданной позицией
в контексте, дающая новый контекст, в котором этот тип
заменён на $t$ с надлежащим индексом.
\end{definition}

\begin{definition}[Сжатие до одного элемента]
Извлечение одного типа из контекста длины $S\,(m+n)$
путём снятия первых $m$ типов.
\end{definition}

\begin{lemma}
Применение переименователя ослабления к преемнику
индекса равно преемнику применения переименователя
ослабления к индексу.
\end{lemma}

\begin{lemma}
Ослабление переименователя ослабления равно
самому переименователю ослабления на преемниках индексов.
\end{lemma}

\begin{lemma}
Переименование по ослаблению переименователя ослабления
равносильно переименованию по переименователю ослабления.
\end{lemma}

\begin{lemma}[Преемник индекса]
Индексирование контекста, расширенного новым типом,
по преемнику индекса даёт сдвиг индекса в исходном
контексте.
\end{lemma}

\begin{lemma}[Вставка и индексирование]
Индексирование вставленного контекста по ослабленному
индексу даёт переименование исходного индекса.
\end{lemma}

\section{Файл \texttt{Levels.v}}

\begin{class}[Сигнатура уровней]
\emph{Сигнатура уровней} на типе $L$ состоит из
бинарного отношения $\ax$ (представляющего, что один
уровень является аксиомой на другом уровне) и
трёхместного отношения $\ruleL$ (представляющего
формирование типа произведения из уровней области
и области значений).

\begin{class}[Функциональная сигнатура уровней]

каждый уровень имеет аксиому (для каждого $\ell$
существует $\ell'$ такой, что $\ax(\ell, \ell')$);
и каждая пара уровней определяет уровень произведения
(для каждого $\ell_t, \ell_u$ существует $\ell$ такой,
что $\ruleL(\ell_t, \ell_u, \ell)$).
\end{class}

\section{Файл \texttt{Typing.v}}
%==============================================================================

В данном файле определяется суждение типирования
и доказаны ключевые метатеоретические леммы
ослабления и подстановки.

Файл импортирует стандартную библиотеку, а также модули
контекстов, тождественного равенства, уровней и синтаксиса.

\begin{definition}[Суждение типирования]
Суждение $\Gamma \vdash t \Leftarrow T$
(читается: «в контексте $\Gamma$ терм $t$ имеет тип $T$»)
определяется индуктивно следующими правилами:
\begin{itemize}[leftmargin=2em,itemsep=2pt]
\item \textbf{Вселенная:} Если $\ax(\ell, \ell')$,
то $\Gamma \vdash \UU(\ell) \Leftarrow \UU(\ell')$.
\item \textbf{Произведение:} Если
$\Gamma \vdash T \Leftarrow \UU(\ell_t)$,
$\Gamma\;\&\;T \vdash U \Leftarrow \UU(\ell_u)$
и $\ruleL(\ell_t, \ell_u, \ell)$, то
$\Gamma \vdash \PiT\,T\;U \Leftarrow \UU(\ell)$.
\item \textbf{Переменная:}
$\Gamma \vdash \var(i) \Leftarrow \Gamma \mathbin{!!} i$.
\item \textbf{Лямбда:} Если
$\Gamma \vdash U \Leftarrow \UU(\ell)$
и $\Gamma\;\&\;U \vdash t \Leftarrow T$, то
$\Gamma \vdash \lam\,U\;t \Leftarrow \PiT\,U\;T$.
\item \textbf{Применение:} Если
$\Gamma \vdash t \Leftarrow \PiT\,U\;T$
и $\Gamma \vdash u \Leftarrow U$, то
$\Gamma \vdash t\;\app\;u \Leftarrow T \sub u$.
\item \textbf{Конверсия:} Если
$\Gamma \vdash t \Leftarrow T$,
$\Gamma \vdash T' \Leftarrow \UU(\ell)$
и $T \Deq T'$, то
$\Gamma \vdash t \Leftarrow T'$.
\end{itemize}
\end{definition}

\begin{lemma}[Сильное ослабление]
Если $\Gamma \vdash t \Leftarrow T$ и мы вставляем новый
тип $U$ в начало контекста, то переименованный терм
и тип остаются корректно типированными:
$\insertt(\Gamma, U) \vdash \rename(\wrr)(t) \Leftarrow \rename(\wrr)(T)$.
\end{lemma}

\begin{lemma}[Ослабление]
Если $\Gamma \vdash t \Leftarrow T$, то для любого типа $U$
$\Gamma\;\&\;U \vdash \shift(t) \Leftarrow \shift(T)$.
\end{lemma}

\begin{lemma}[Сильная подстановка]
Если $\Gamma \vdash t \Leftarrow T$
и $\shrink(\Gamma) \vdash u \Leftarrow \squeeze(\Gamma)$,
то подстановка $u$ в контекст и терм/тип сохраняет
суждение типирования.
\end{lemma}

\begin{lemma}[Подстановка]
Если $\Gamma\;\&\;U \vdash t \Leftarrow T$
и $\Gamma \vdash u \Leftarrow U$, то
$\Gamma \vdash t \sub u \Leftarrow T \sub u$.
\end{lemma}

\section{Файл \texttt{Inversions.v}}
%==============================================================================

В данном файле доказываются леммы инверсии,
восстанавливающие структуру вывода типирования
из его заключения.

Файл импортирует стандартную библиотеку, а также модули
абстрактного переписывания, тождественного равенства,
уровней, типирования, контекстов, синтаксиса, конгруэнтности
и редукции.

\begin{lemma}[Инверсия вселенной]
Если $\Gamma \vdash \UU(\ell) \Leftarrow T$, то существует
уровень $\ell'$ такой, что $\ax(\ell, \ell')$
и $T \Deq \UU(\ell')$.
\end{lemma}

\begin{lemma}[Инверсия произведения]
Если $\Gamma \vdash \PiT\,T\;U \Leftarrow V$, то
существуют уровни $\ell_t, \ell_u, \ell$ такие, что
$\ruleL(\ell_t, \ell_u, \ell)$,
$\Gamma \vdash T \Leftarrow \UU(\ell_t)$,
$\Gamma\;\&\;T \vdash U \Leftarrow \UU(\ell_u)$
и $V \Deq \UU(\ell)$.
\end{lemma}

\begin{lemma}[Инверсия переменной]
Если $\Gamma \vdash \var(i) \Leftarrow T$, то
$T \Deq \Gamma \mathbin{!!} i$.
\end{lemma}

\begin{lemma}[Инверсия лямбды]
Если $\Gamma \vdash \lam\,T\;t \Leftarrow V$, то
существуют уровень $\ell$ и терм $U$ такие, что
$\Gamma \vdash T \Leftarrow \UU(\ell)$,
$\Gamma\;\&\;T \vdash t \Leftarrow U$
и $V \Deq \PiT\,T\;U$.
\end{lemma}

\begin{lemma}[Инверсия применения]
Если $\Gamma \vdash t\;\app\;u \Leftarrow V$, то
существуют термы $U$ и $T$ такие, что
$\Gamma \vdash t \Leftarrow \PiT\,U\;T$,
$\Gamma \vdash u \Leftarrow U$
и $V \Deq T \sub u$.
\end{lemma}

\begin{theorem}[Единственность типирования / Единственность типа]
Если сигнатура уровней функциональна, и
$\Gamma \vdash t \Leftarrow T$ и
$\Gamma \vdash t \Leftarrow T'$, то $T \Deq T'$.
Иными словами, терм имеет не более одного типа
с точностью до тождественного равенства.
\end{theorem}

\section{Файл \texttt{WF.v}}
%==============================================================================

В данном файле определяется хорошо определённость контекстов.

Файл импортирует модули уровней, контекстов, типирования
и синтаксиса.

\begin{definition}[Хорошо определённые контексты]
Контекст $\Gamma$ \emph{хорошо определён}
(обозначение $\wf(\Gamma)$), если:
пустой контекст хорошо определён; и если $\Gamma$
хорошо определён и
$\Gamma \vdash T \Leftarrow \UU(\ell)$ для некоторого
уровня $\ell$, то расширение
$\Gamma\;\&\;T$ хорошо определено.
\end{definition}

\section{Файл \texttt{Metatheory.v}}
%==============================================================================

В данном файле доказывается свойство сохранения типов
(\emph{subject reduction}).

Файл импортирует стандартную библиотеку, а также модули
типирования, контекстов, хорошо определённости, уровней,
синтаксиса, тождественного равенства, редукции,
абстрактного переписывания, инверсий и конгруэнтности.

\begin{lemma}[Сильная редукция контекста]
Если $\Gamma \vdash t \Leftarrow T$, $\Gamma$ хорошо
определён, и $\Gamma \dstep \Delta$ (шаг
$\beta$-преобразования в контексте), то
$\Delta \vdash t \Leftarrow T$.
Иными словами, редукция типа в контексте
сохраняет типирование терма.
\end{lemma}

\begin{lemma}[Редукция контекста]
Если $\Gamma\;\&\;U \vdash t \Leftarrow T$, $\Gamma$
хорошо определён, $\Gamma \vdash U \Leftarrow \UU(\ell)$
и $U \bstep U'$, то
$\Gamma\;\&\;U' \vdash t \Leftarrow T$.
Иными словами, редукция типа в начале контекста
сохраняет типирование терма в расширенном контексте.
\end{lemma}

\begin{lemma}[Корректность типа]
Если сигнатура уровней полна, и
$\Gamma \vdash t \Leftarrow T$ при том, что $\Gamma$
хорошо определён, то существует уровень $\ell$ такой,
что $\Gamma \vdash T \Leftarrow \UU(\ell)$.
Иными словами, тип в суждении типирования
сам является хорошо типированным как вселенная.
\end{lemma}

\begin{theorem}[Сохранение типов]
Если сигнатура уровней полна, и
$\Gamma \vdash t \Leftarrow T$ при том, что $\Gamma$
хорошо определён, и $t \bstep t'$, то
$\Gamma \vdash t' \Leftarrow T$.
Иными словами, $\beta$-преобразование сохраняет
типирование: если терм имеет тип, то его
$\beta$-редукт также имеет тот же тип.
\end{theorem}

\section{Файл \texttt{Atomic.v}}
%==============================================================================

В данном файле определяется понятие атомарного терма
и доказывается сохранение атомарности при
переименовании и $\beta$-преобразовании.

Файл импортирует стандартную библиотеку, а также
модули синтаксиса и редукции.

\begin{definition}[Атомарные термы]
Терм \emph{атомарен}, если он есть: вселенная;
произведение; переменная; или применение атомарной
функции к атомарному аргументу.
\end{definition}

\begin{lemma}[Атомарность сохраняется при переименовании]
Если терм атомарен, то любое его переименование
также атомарно.
\end{lemma}

\begin{lemma}[Атомарность сохраняется при $\beta$-преобразовании]
Если $t \bstep t'$ и $t$ атомарен, то $t'$
также атомарен.
\end{lemma}

\section{Файл \texttt{StronglyNormalizing.v}}
%==============================================================================

В данном файле определяется сильная нормализуемость
и доказываются ряд вспомогательных лемм.

Файл импортирует стандартную библиотеку, а также
модули синтаксиса, редукции и атомарности.

\begin{definition}[Доступность / Обоснованность]
Для отношения $R$ на множестве $A$ элемент $x$
\emph{доступен} (относительно $R$), если каждый
$y$ такой, что $x\;R\;y$, сам доступен.
\end{definition}

\begin{definition}[Сильная нормализуемость]
Терм $t$ \emph{сильно нормализуем}
(обозначение $\SN(t)$), если он доступен
относительно отношения одношагового
$\beta$-преобразования.
\end{definition}

\begin{lemma}[Сильная нормализуемость по голове]
Если $t\;\app\;u$ сильно нормализуем, то $t$
сильно нормализуем.
\end{lemma}

\begin{lemma}[Сильная нормализуемость атомарного применения]
Если $t$ атомарен и сильно нормализуем,
и $u$ сильно нормализуем, то $t\;\app\;u$
сильно нормализуем.
\end{lemma}

\begin{lemma}[Переименование сохраняет сильную нормализуемость]
Если терм сильно нормализуем, то любое его
переименование также сильно нормализуемо.
\end{lemma}

\begin{definition}[Слабое головное расширение]
Отношение $\WHE$ (слабое головное расширение)
на термах определяется: если $t$ слабо головно
расширяется в $t'$, то $t\;\app\;u$ слабо головно
расширяется в $t'\;\app\;u$; и если $U$ и $u$
сильно нормализуемы, то
$\lam\,U\;t\;\app\;u$ слабо головно расширяется
в $t \sub u$.
\end{definition}

\begin{definition}[Аджаст]
Функция, которая для заданного $m$ отображает индекс
в $\Fin(n)$ в индекс в $\Fin(m+n)$ путём сдвига
всех индексов на $m$.
\end{definition}

\begin{lemma}[Аджаст и сильная нормализуемость]
Если терм, переименованный функцией $\adjust$,
сильно нормализуем, то исходный терм
сильно нормализуем.
\end{lemma}

\begin{lemma}[Переименование коммутирует со слабым головным расширением]
Если $t$ слабо головно расширяется в $t'$,
то любое переименование $t$ слабо головно
расширяется в переименование $t'$.
\end{lemma}

\begin{lemma}[Слабое головное расширение и сильная нормализуемость]
Если $t$ слабо головно расширяется в $t'$
и $t'\;\app\;u$ сильно нормализуем, то
$t\;\app\;u$ сильно нормализуем.
\emph{Эта лемма объявлена аксиомой в ожидании
исправления на основе доказательства Альтенкирха.}
\end{lemma}
